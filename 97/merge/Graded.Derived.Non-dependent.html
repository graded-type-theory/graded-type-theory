private variable
  A B : Term _
  b   : BinderMode
  γ δ : Conₘ _
  m   : Mode
  p   : M

opaque
  unfolding _⟶×⟨_⟩[_]_

  -- A usage rule for _⟶×⟨_⟩[_]_.

  ▸⟶× :
    γ ▸[ m ᵐ· p ] A →
    δ ▸[ m ] B →
    p ·ᶜ γ +ᶜ δ ▸[ m ] A ⟶×⟨ b ⟩[ p ] B
  ▸⟶× {m} {δ} ▸A ▸B =
    ΠΣₘ ▸A $
    sub (wkUsage _ ▸B) $ begin
      δ ∙ ⌜ m ⌝ · 𝟘  ≈⟨ ≈ᶜ-refl ∙ ·-zeroʳ _ ⟩
      δ ∙ 𝟘          ∎

opaque
  unfolding _⟶×⟨_⟩[_]_

  -- An inversion lemma for _⟶×⟨_⟩[_]_.

  inv-usage-⟶× :
    γ ▸[ m ] A ⟶×⟨ b ⟩[ p ] B →
    ∃₂ λ δ η →
      γ ≤ᶜ p ·ᶜ δ +ᶜ η ×
      δ ▸[ m ᵐ· p ] A ×
      η ▸[ m ] B
  inv-usage-⟶× ▸⟶× =
    let invUsageΠΣ ▸A ▸B γ≤ = inv-usage-ΠΣ ▸⟶× in
    _ , _ , γ≤ , ▸A , wkUsage⁻¹ ▸B
