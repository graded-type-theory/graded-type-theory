opaque

  -- Validity of Quot.

  Quotʳ :
    ts » Γ ⊢ l ∷Level →
    γ ▸ Γ ⊩ʳ Quot A B ∷[ m ∣ n ] U l
  Quotʳ ⊢l =
    ▸⊩ʳ∷⇔ .proj₂ λ ⊢σ _ →
    ®∷→®∷◂ $
    ®∷U⇔ .proj₂
      ( subst-⊢ ⊢l ⊢σ
      , U/Levelᵣ (λ { PE.refl → T.refl })
      )

opaque

  -- Validity of class.

  classʳ :
    ts » Γ ⊢ Quot A B →
    ts » Γ ⊢ t ∷ A →
    γ ▸ Γ ⊩ʳ t ∷[ m ∣ n ] A →
    γ ▸ Γ ⊩ʳ class t ∷[ m ∣ n ] Quot A B
  classʳ {m = 𝟘ᵐ} _ _ _ =
    ▸⊩ʳ∷[𝟘ᵐ]
  classʳ {m = 𝟙ᵐ} ⊢Q ⊢t ⊩ʳt =
    ▸⊩ʳ∷⇔ .proj₂ λ ⊢σ σ®σ′ →
    let ⊢t[σ] = subst-⊢ ⊢t ⊢σ in
    ®∷→®∷◂ $
    ®∷Quot⇔ .proj₂
      ( _
      , ⇒*→⇛ (id (class (subst-⊢ ⊢Q ⊢σ) ⊢t[σ]))
      , ®∷→®∷◂ω non-trivial (▸⊩ʳ∷⇔ .proj₁ ⊩ʳt ⊢σ σ®σ′)
      )

opaque

  -- Validity of resp.

  respʳ :
    Higher-quotient-constructors-allowed →
    m PE.≡ 𝟘ᵐ? →
    γ ▸ Γ ⊩ʳ resp A B t u v ∷[ m ∣ n ] C
  respʳ ok PE.refl =
    ▸⊩ʳ∷[𝟘ᵐ?] ok

opaque

  -- Validity of set.

  setʳ :
    Higher-quotient-constructors-allowed →
    m PE.≡ 𝟘ᵐ? →
    γ ▸ Γ ⊩ʳ set A B t u v w ∷[ m ∣ n ] C
  setʳ ok PE.refl =
    ▸⊩ʳ∷[𝟘ᵐ?] ok
